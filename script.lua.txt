local Runtime = {Active = true, Connections = {}, Instances = {}, Tweens = {}, Workers = {}, Cleanups = {}}
function Runtime:TrackConnection(conn)
    if conn then table.insert(self.Connections, conn) end
    return conn
end
function Runtime:TrackInstance(instance)
    if instance then table.insert(self.Instances, instance) end
    return instance
end
function Runtime:TrackTween(tween)
    if tween then table.insert(self.Tweens, tween) end
    return tween
end
function Runtime:EventCallback(callback)
    return function(...)
        if self.Active then
            local ok, err = pcall(callback, ...)
            if not ok then warn("[klz bypasse] " .. tostring(err)) end
            if not ok then warn("[zenith store] " .. tostring(err)) end
        end
    end
end
function Runtime:Report(name, err)
    warn("[klz bypasse] " .. tostring(name) .. ": " .. tostring(err))
    warn("[zenith store] " .. tostring(name) .. ": " .. tostring(err))
end
function Runtime:RunUnique(name, callback)
    local old = self.Workers[name]
    if old and coroutine.status(old) ~= "dead" then return old end
    local thread = task.spawn(function()
        local ok, err = pcall(callback)
        if not ok then warn("[klz bypasse] " .. name .. ": " .. tostring(err)) end
        if not ok then warn("[zenith store] " .. name .. ": " .. tostring(err)) end
    end)
    self.Workers[name] = thread
    return thread
end
function Runtime:HookMetamethod(target, method, callback)
    local old
    local function dispatch(object, ...)
        if not self.Active then return old(object, ...) end
        return callback(object, ...)
    end
    local replacement = type(newcclosure) == "function" and newcclosure(dispatch) or dispatch
    old = hookmetamethod(target, method, replacement)
    table.insert(self.Cleanups, function()
        if type(getrawmetatable) == "function" then
            local mt = getrawmetatable(target)
            if mt and mt[method] == replacement then hookmetamethod(target, method, old) end
        end
    end)
    return old
end

local anti = {running = false, originals = setmetatable({}, {__mode = "k"})}
local function patchAntiFly()
    if game.PlaceId ~= 9417205094 or type(filtergc) ~= "function" then return false end
    if type(debug) ~= "table" or type(debug.getinfo) ~= "function"
        or type(debug.getupvalue) ~= "function" or type(debug.setupvalue) ~= "function" then return false end
    local character = LocalPlayer.Character
    local scripts = LocalPlayer:FindFirstChild("PlayerScripts")
    local target = scripts and scripts:FindFirstChild("ChatService")
    if not character or not target then return false end
    local ok, fn = pcall(filtergc, "function", {
        Name = "antiflyfunc", StartLine = 541, Upvalues = {character},
    }, true)
    if not ok or not fn then return false end
    local infoOk, info = pcall(debug.getinfo, fn)
    if not infoOk or not info or info.source ~= "=" .. target:GetFullName() then return false end
    local valueOk, value = pcall(debug.getupvalue, fn, 3)
    if not valueOk or type(value) ~= "number" then return false end
    if anti.originals[fn] == nil then anti.originals[fn] = value end
    return pcall(debug.setupvalue, fn, 3, -1000000000)
end
local function stopAntiFly()
    anti.running = false
    for fn, original in pairs(anti.originals) do
        pcall(debug.setupvalue, fn, 3, original)
        anti.originals[fn] = nil
    end
end
local function startAntiFly()
    if anti.running then return end
    anti.running = true
    Runtime:RunUnique("KLZAntiFly", function()
        while Runtime.Active and anti.running do
            patchAntiFly()
            task.wait(0.75)
        end
    end)
end

local function SimularCliqueUI(botao)
    local sucesso = false
    
    -- Tenta via getconnections
    if getconnections then
        pcall(function()
            local eventos = {"MouseButton1Click", "MouseButton1Down", "MouseButton1Up", "Activated"}
            for _, nomeEvento in ipairs(eventos) do
                for _, conn in ipairs(getconnections(botao[nomeEvento])) do
                    conn:Fire()
                    sucesso = true
                end
            end
        end)
    end
    
    -- Fallback: firesignal
    if firesignal then
        pcall(function() firesignal(botao.MouseButton1Down) end)
        pcall(function() firesignal(botao.MouseButton1Click) end)
        pcall(function() firesignal(botao.MouseButton1Up) end)
        pcall(function() firesignal(botao.Activated) end)
        sucesso = true
    end
    
    -- Fallback: VirtualInputManager (Teclado/Controle)
    -- Como você está no celular, o botão de pulo (Mobile2) fica na frente da tela e absorve o clique do mouse.
    -- Para burlar isso, nós selecionamos o botão diretamente e apertamos "Enter" / "Botão A".
    pcall(function()
        local antigoSelecionado = GuiService.SelectedObject
        GuiService.SelectedObject = botao
        task.wait(0.1)
        
        -- Tenta clicar com Enter (Teclado)
        VirtualInputManager:SendKeyEvent(true, Enum.KeyCode.Return, false, game)
        task.wait(0.05)
        VirtualInputManager:SendKeyEvent(false, Enum.KeyCode.Return, false, game)
        
        -- Tenta clicar com Botão A (Controle do Xbox)
        VirtualInputManager:SendKeyEvent(true, Enum.KeyCode.ButtonA, false, game)
        task.wait(0.05)
        VirtualInputManager:SendKeyEvent(false, Enum.KeyCode.ButtonA, false, game)
        
        GuiService.SelectedObject = antigoSelecionado
    end)
    sucesso = true
    
    if not sucesso then
        warn("Seu executor não suporta nenhum método de clique.")
    end
end

local function findVehicle()
    local folder = workspace:FindFirstChild("CarrosSpawnados")
    if not folder then return nil end
    for _, v in ipairs(folder:GetChildren()) do
        local info = v:FindFirstChild("Info")
        if info then
            local owner = info:FindFirstChild("Owner")
            if owner and owner.Value == playerName then
                return v
            end
        end
    end
    return nil
end

local function EsperarCarro()
    local inicio = tick()
    while tick() - inicio < CONFIG.TEMPO_ESPERA_CARRO do
        local v = findVehicle()
        if v then return v end
        task.wait(0.3)
    end
    return nil
end

local function AndarAteCarro(vehicle)
    if not vehicle then return false end
    local alvoPart = vehicle.PrimaryPart or vehicle:FindFirstChildWhichIsA("BasePart")
    if not alvoPart then return false end

    
    local distInicial = (HRP.Position - alvoPart.Position).Magnitude
    if distInicial > 5 then
        local dir = (HRP.Position - alvoPart.Position).Unit
        local destinoFly = alvoPart.Position + (dir * 5)
        -- Para o TpToFly aceitar Vector3, ele ja converte internamente
        TpToFly(destinoFly)
        task.wait(0.5)
    end
    
    local inicio = tick()
    local ultimaPos = HRP.Position
    local tempoParado = 0
    
    while tick() - inicio < CONFIG.TEMPO_MAX_ANDAR do
        local destino = alvoPart.Position
        local dist = (HRP.Position - destino).Magnitude
        
        if dist <= CONFIG.DISTANCIA_CHEGADA then
            Hum:MoveTo(HRP.Position)
            return true
        end
        
        -- Verifica se esta travado em parede
        local moveu = (HRP.Position - ultimaPos).Magnitude
        if moveu < 0.5 then
            tempoParado = tempoParado + 0.1
            if tempoParado >= 1 then
                Hum.Jump = true
                task.wait(0.3)
                tempoParado = 0
            end
        else
            tempoParado = 0
        end
        
        ultimaPos = HRP.Position
        Hum:MoveTo(destino)
        task.wait(0.1)
    end
    
    warn("Falha ao chegar no carro (Timeout)")
    return false
end

local function EntrarNoCarro(vehicle)
    if not vehicle then return false end
    
    local body = vehicle:FindFirstChild("Body") or vehicle:FindFirstChild("Body", true)
    if not body then return false end
    
    local ppDriveSeat = body:FindFirstChild("PPDriveSeat")
    if not ppDriveSeat then return false end
    
    local prompt = ppDriveSeat:FindFirstChildOfClass("ProximityPrompt")

    for i = 1, 10 do

            local seat = Hum.SeatPart
            if seat and seat:IsDescendantOf(vehicle) then
                return true
            end
        end
    end
    
    warn("⚠️ Não conseguiu entrar no carro")
    return false
end

local function ZerarVelocidadeCarro(partes)

end

local function TeleportarCarro(vehicle, destinoCFrame)
    if not vehicle then return false end
    local pPart = vehicle.PrimaryPart or vehicle:FindFirstChildWhichIsA("BasePart")
    if not pPart then return false end

    -- Se for um Vector3 converte para CFrame
    if typeof(destinoCFrame) == "Vector3" then
        destinoCFrame = CFrame.new(destinoCFrame)
    end
    local alvo = destinoCFrame.Position

    local partes = {}
    for _, d in ipairs(vehicle:GetDescendants()) do
        if d:IsA("BasePart") then table.insert(partes, d) end
    end

    for tentativa = 1, 5 do
        if not rodando then return false end
        if not Hum.Sit then
            warn("Saiu do carro durante o teleporte!")
            return false
        end

        -- TP direto do minerador: mantem o carro no destino por ~1s.
        local t0 = tick()
        while tick() - t0 < 1 do
            local seat = Hum.SeatPart
            if not rodando or not Hum.Sit or not seat or not seat:IsDescendantOf(vehicle) then
                warn("Jogador saiu do carro durante o teleporte.")
                return false
            end
            pcall(function() vehicle:PivotTo(destinoCFrame) end)
            ZerarVelocidadeCarro(partes)
            task.wait()
        end

        -- Espera um pouco e confere se o carro NAO voltou
        task.wait(0.8)
        local dist = (pPart.Position - alvo).Magnitude
        if dist < 25 then
            task.wait(0.5) -- Carrega/sincroniza o destino antes de liberar o desembarque.
            local seat = Hum.SeatPart
            local carroNoDestino = vehicle.Parent and (pPart.Position - alvo).Magnitude < 25
            local playerNoCarro = HRP and HRP.Parent and (HRP.Position - pPart.Position).Magnitude < 35
            if rodando and carroNoDestino and playerNoCarro and seat and seat:IsDescendantOf(vehicle) then
                return true
            end
        end
        -- Voltou: teleporta de novo SEM sair do carro (evita pular e entrar de novo)
        warn(("Carro voltou (%.0f studs do destino). Teleportando de novo..."):format(dist))
    end

    warn("Carro nao estabilizou no destino apos 5 tentativas.")
    return false
end

local function PularDoCarro(vehicle)
    for tentativa = 1, 10 do
        pcall(function()
            Hum.Sit = false
            Hum:ChangeState(Enum.HumanoidStateType.Jumping)
        end)
        task.wait(0.15)

        if not Hum.Sit and not Hum.SeatPart then
            -- Afasta o player do banco pra ele nao sentar sozinho de novo ao encostar
            local pPart = vehicle and (vehicle.PrimaryPart or vehicle:FindFirstChildWhichIsA("BasePart"))
            if pPart then
                pcall(function()
                    HRP.CFrame = pPart.CFrame * CFrame.new(8, 3, 0)
                    HRP.AssemblyLinearVelocity = Vector3.zero
                end)
            end
            task.wait(0.5)
            if not Hum.Sit and not Hum.SeatPart then
                return true
            end
            warn("Sentou de novo sozinho, saindo outra vez...")
        end
    end
    warn("Nao conseguiu sair do carro!")
    return false
end

-- Função rápida que faz todo o ciclo de ir e entrar no carro
local function PegarCarroETeleportar(destinoCFrame)
    SpawnarCarro()
    local veiculo = EsperarCarro()
    if not veiculo then
        warn("Carro não spawnou!")
        return false
    end
    
    if not AndarAteCarro(veiculo) then return false end
    if not EntrarNoCarro(veiculo) then return false end
    
    task.wait(0.5) -- espera o assento sincronizar antes do TP
    local chegou = TeleportarCarro(veiculo, destinoCFrame)
    if not chegou or not rodando then return false end
    task.wait(0.5) -- Pausa no destino antes de sair e continuar a tarefa.
    if not rodando then return false end
    if not PularDoCarro(veiculo) then return false end
    return veiculo
end

local function VerificarFomeSede()

local function contarEquipados(nomeItem)
    local inventario = LocalPlayer:FindFirstChild("Inventario")
    if not inventario then return 0 end
    local count = 0
    for _, obj in ipairs(inventario:GetChildren()) do
        if obj:IsA("BoolValue") and obj.Name == nomeItem and obj.Value == true then
            count = count + 1
        end
    end
    return count
end

local function pegarUidNaoEquipado(nomeItem)
    local invGui = abrirInventario()
    if not invGui then return nil end
    local front = invGui:FindFirstChild("ToolStorage") and invGui.ToolStorage:FindFirstChild("Front")
    local bauFrame = front and front:FindFirstChild("BauFrame")
    if not bauFrame then return nil end
    
    for _, slot in ipairs(bauFrame:GetChildren()) do
        if slot.Name:find("Slot") then
            local armaName = slot:FindFirstChild("ArmaName")
            local uidValue = slot:FindFirstChild("Uid")
            local equipado = slot:FindFirstChild("Equipado")
            
            if armaName and armaName.Value == nomeItem and uidValue and uidValue.Value ~= "" then
                if equipado and equipado.Value == false then
                    return uidValue.Value
                end
            end
        end
    end
    return nil
end

local function DesequiparTudo()
    local invGui = abrirInventario()
    if not invGui then return end
    
    local front = invGui:FindFirstChild("ToolStorage") and invGui.ToolStorage:FindFirstChild("Front")
    local equipedFrame = front and front:FindFirstChild("EquipedFrame")
    if not equipedFrame then return end
    
    for _, slot in ipairs(equipedFrame:GetChildren()) do
        if slot.Name:find("Slot") then
            local ocupado = slot:FindFirstChild("Ocupado")
            local nome = slot:FindFirstChild("ArmaName")
            local uid = slot:FindFirstChild("Uid")
            
            if ocupado and ocupado.Value == true and nome and uid and uid.Value ~= "" then
                local args = {nome.Value, "Desequipar", uid.Value}
                game:GetService("ReplicatedStorage"):WaitForChild("InventarioSystem"):WaitForChild("Equipar"):FireServer(unpack(args))
                task.wait(0.2)
            end
        end
    end
end

local function GarantirItensEComprar(nomeItem, quantidade)
    
    -- Tenta equipar o que tem
    local uidsUsados = {}
    for i = 1, quantidade do
        local uid = nil
        local invGui = abrirInventario()
        local front = invGui and invGui:FindFirstChild("ToolStorage") and invGui.ToolStorage:FindFirstChild("Front")
        local bauFrame = front and front:FindFirstChild("BauFrame")
        
        if bauFrame then
            for _, slot in ipairs(bauFrame:GetChildren()) do
                if slot.Name:find("Slot") then
                    local sNome = slot:FindFirstChild("ArmaName")
                    local sUid = slot:FindFirstChild("Uid")
                    if sNome and sNome.Value == nomeItem and sUid and sUid.Value ~= "" and not uidsUsados[sUid.Value] then
                        uid = sUid.Value
                        break
                    end
                end
            end
        end
        if uid then
            uidsUsados[uid] = true
            local args = {nomeItem, "Equipar", uid}
            game:GetService("ReplicatedStorage"):WaitForChild("InventarioSystem"):WaitForChild("Equipar"):FireServer(unpack(args))
            task.wait(CONFIG.DELAY_EQUIPAR)
        end
    end
    
    local equipados = contarEquipados(nomeItem)
    if equipados < quantidade then
        local falta = quantidade - equipados
        for i = 1, falta do
            local args = {nomeItem}
            game:GetService("ReplicatedStorage"):WaitForChild("InventarioSystem"):WaitForChild("Comprar"):FireServer(unpack(args))
            task.wait(CONFIG.DELAY_COMPRA)
        end
        task.wait(CONFIG.DELAY_VERIFICAR)
        
        for i = 1, falta do
            local nUid = pegarUidNaoEquipado(nomeItem)
            if nUid then
                local args = {nomeItem, "Equipar", nUid}
                game:GetService("ReplicatedStorage"):WaitForChild("InventarioSystem"):WaitForChild("Equipar"):FireServer(unpack(args))
                task.wait(CONFIG.DELAY_EQUIPAR)
            end
        end
    end
end

local function ConsumirItem(nomeItem, quantidade)
    if not ComerRemote then return end
    
    local char = LocalPlayer.Character
    if not char then return end
    
    for i = 1, quantidade do
        -- Tenta equipar simulando as teclas 1, 2, 3 com delay (Lógica do Auto Erva)
        local equipado = false
        if char:FindFirstChild(nomeItem) then
            equipado = true
        else
            for slot = 1, 3 do
                PressNumberKey(slot)
                task.wait(0.3)
                if char:FindFirstChild(nomeItem) then
                    equipado = true
                    break
                end
            end
        end
        
        if equipado then
            local args = {nomeItem}; game:GetService("ReplicatedStorage"):WaitForChild("ComerComida"):FireServer(unpack(args))
            task.wait(CONFIG.DELAY_COMER)
        else
            warn("⚠️ Não foi possível equipar a comida: " .. nomeItem)
        end
    end
end

    end
    
    -- Compra e Come
    local itensUso = {}
    if sedeBaixa then table.insert(itensUso, "Coca-Cola") end
    if fomeBaixa then table.insert(itensUso, "Pizza") end
    
    for _, itemNome in ipairs(itensUso) do
        local qnts = 3
        for _, v in ipairs(CONFIG.ITENS_COMIDA) do
            if v.nome == itemNome then qnts = v.quantidade break end
        end
        GarantirItensEComprar(itemNome, qnts)
        ConsumirItem(itemNome, qnts)
    end
    
    -- Como a base de mineração já é nosso trabalho atual, pegamos o carro de novo e teleportamos para o Minerador (ou pra posição exata salva)
    -- O mais seguro é teleportar pro ponto base do minerador, pra não bugar na parede/pedra
    if not PegarCarroETeleportar(CONFIG.WP_MINERADOR) then
        -- Se falhar o carro, teleporta a pé mesmo
        warn("Retorno do veiculo nao confirmado; nova tentativa no proximo ciclo.")
        return false
    end
    
end

    end

local TweenService = game:GetService("TweenService")

TpToFly = function(destino)
    if typeof(destino) == "CFrame" then
        destino = destino
    else
        destino = CFrame.new(destino)
    end

    local char = LocalPlayer.Character
    if not char then return end
    local hrp = char:FindFirstChild("HumanoidRootPart")
    if not hrp then return end

    local dist = (hrp.Position - destino.Position).Magnitude
    local speed = 80 -- TpTo original do minerador, com velocidade maior (antes 39).
    local tempo = math.max(dist / speed, 0.3)

    local bv = Runtime:TrackInstance(Instance.new("BodyVelocity"))
    bv.Velocity = Vector3.new(0, 0, 0)
    bv.MaxForce = Vector3.new(math.huge, math.huge, math.huge)
    bv.Parent = hrp

    local tweenInfo = TweenInfo.new(tempo, Enum.EasingStyle.Linear)
    local tween = Runtime:TrackTween(TweenService:Create(hrp, tweenInfo, {CFrame = destino}))
    tween:Play()
    tween.Completed:Wait()

    if bv then bv:Destroy() end
end

local pedrasBugadas = {

local function AcharPedraMaisProxima(pedrasFolder, pedrasJaMineradas)
    local hrp = LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("HumanoidRootPart")
    if not hrp then return nil end
    
    local maisProxima = nil
    local menorDist = math.huge
    
    for _, pedra in ipairs(pedrasFolder:GetChildren()) do
        if not pedrasJaMineradas[pedra] then
            local base = pedra:FindFirstChildWhichIsA("BasePart") or (pedra:IsA("BasePart") and pedra)
            if base and base.CanCollide == true then
                -- Ignora as pedras bugadas baseando-se na posicao
                local ehBugada = false
                for _, posBugada in ipairs(pedrasBugadas) do
                    if (base.Position - posBugada).Magnitude < 5 then
                        ehBugada = true
                        break
                    end
                end
                
                if not ehBugada then
                    local dist = (base.Position - hrp.Position).Magnitude
                    if dist < menorDist then
                        menorDist = dist
                        maisProxima = pedra
                    end
                end
            end
        end
    end
    
    return maisProxima
end

local function IrParaDestino(destinoCFrame)
    local dist = (HRP.Position - destinoCFrame.Position).Magnitude
    if dist > 100 then
        local carroPego = false
        local tentativas = 0
        while rodando and not carroPego and tentativas < 3 do
            carroPego = PegarCarroETeleportar(destinoCFrame)
            if not carroPego then
                tentativas = tentativas + 1
                warn("Falhou ao usar carro, tentando novamente...")
                task.wait(2)
            end
        end
        -- Se falhar 3x, aborta para n�o ir voando de jeito nenhum
        if not carroPego and rodando then
            warn("Carro falhou 3x. Abortando TP para nao voar.")
            return false
        end
    else
        if Hum.Sit then PularDoCarro() end
        TpToFly(destinoCFrame)
    end
end

local function PegarPicareta()
    local npcCoord = CFrame.new(401.166992, 5.32570934, 2114.17896)
    
    IrParaDestino(npcCoord + Vector3.new(0, 3, 3))
    task.wait(1)
    
    local promptEncontrado = nil
    for _, obj in pairs(workspace:GetDescendants()) do
        if obj:IsA("ProximityPrompt") then
            local part = obj.Parent
            if part and part:IsA("BasePart") then
                if (part.Position - npcCoord.Position).Magnitude < 15 then
                    promptEncontrado = obj
                    break
                end
            end
        end
    end
    
    if promptEncontrado then
        PrepararPrompt(promptEncontrado)
        local key = promptEncontrado.KeyboardKeyCode
        
        -- Dispara tanto o toque de celular quanto a tecla de PC
        pcall(function() promptEncontrado:InputHoldBegan() end)
        pcall(function() VirtualInputManager:SendKeyEvent(true, key, false, game) end)
        
        -- Fica segurando até a barrinha encher + 1 segundo extra de garantia!
        local tempoSegurando = promptEncontrado.HoldDuration + 1
        task.wait(tempoSegurando)
        
        pcall(function() promptEncontrado:InputHoldEnded() end)
        pcall(function() VirtualInputManager:SendKeyEvent(false, key, false, game) end)
        
        -- Espera um pouco antes de sair andando pra dar tempo do servidor processar que pegou
        task.wait(1.5)
        return true
    else
        warn("⚠️ ProximityPrompt do NPC da Picareta não encontrado!")
        return false
    end
end

local function InventarioEstaCheio()
    local inventario = LocalPlayer:FindFirstChild("Inventario")
    if not inventario then return false end

    local pesoAtributo = tonumber(inventario:GetAttribute("Peso"))
    local pesoValue = inventario:FindFirstChild("Peso")
    local pesoPropriedade = pesoValue and tonumber(pesoValue.Value)
    return (pesoAtributo and pesoAtributo >= 81) or (pesoPropriedade and pesoPropriedade >= 81) or false
end
local MINERIOS_INVENTARIO = {
    Bronze = true, Ferro = true, Prata = true, Ouro = true,
    Diamante = true, Meteorito = true,
}

local function ContarMineriosInventario()
    local inventario = LocalPlayer:FindFirstChild("Inventario")
    if not inventario then return 0 end
    local total = 0
    for _, item in ipairs(inventario:GetDescendants()) do
        if MINERIOS_INVENTARIO[item.Name] then
            local quantidade = tonumber(item:GetAttribute("Quantidade"))
            if not quantidade then
                local value = item:FindFirstChild("Quantidade")
                if value and value:IsA("ValueBase") then
                    quantidade = tonumber(value.Value)
                end
            end
            if not quantidade and item:IsA("ValueBase") then
                quantidade = tonumber(item.Value)
            end
            total += math.max(0, quantidade or 1)
        end
    end
    return total
end

local function MinerarPedra(pedra, base, prompt)
    local temNaMao = VerificarPicaretaNaMao()
    if not temNaMao then
        local temNaMochila, toolMochila = VerificarPicaretaNaMochila()
        if temNaMochila and Hum then
            Hum:EquipTool(toolMochila)
            task.wait(1)
        else
            warn("Picareta ausente; buscando no NPC")
            PegarPicareta()
            return "retry"
        end
    end

    if not prompt or not prompt.Parent then return "retry" end
    local antes = ContarMineriosInventario()
    local key = prompt.KeyboardKeyCode
    local inicio = os.clock()
    local minimo = inicio + math.max(1, prompt.HoldDuration + 1)
    local limite = inicio + math.max(16, prompt.HoldDuration + 12)
    local confirmado = false

    pcall(function() prompt:InputHoldBegin() end)
    pcall(function() VirtualInputManager:SendKeyEvent(true, key, false, game) end)
    while rodando and os.clock() < limite do
        local atual = ContarMineriosInventario()
        if atual > antes and os.clock() >= minimo then
            confirmado = true
            break
        end
        task.wait(0.1)
    end
    pcall(function() prompt:InputHoldEnd() end)
    pcall(function() VirtualInputManager:SendKeyEvent(false, key, false, game) end)

    if confirmado then return "ok" end
    if rodando then warn("Minerio nao confirmado no inventario; tentando a pedra novamente") end
    return "retry"
end

local function MinerarTodasAsPedras()
    local construcoes = workspace:FindFirstChild("Construcoes")
    local minerador = construcoes and construcoes:FindFirstChild("Minerador")
    local pedrasFolder = minerador and minerador:FindFirstChild("Pedras")
    if not pedrasFolder then
        warn("Pasta de Pedras nao encontrada!")
        return false
    end

    local pedrasJaMineradas = {}
    while rodando do
        if InventarioEstaCheio() then
            return true
        end

        ChecarEResolverFomeSede()
        if not rodando then return false end

        local pedra = AcharPedraMaisProxima(pedrasFolder, pedrasJaMineradas)
        if not pedra then
            task.wait(2)
            pedrasJaMineradas = {}
            continue
        end

        local base = pedra:FindFirstChildWhichIsA("BasePart") or (pedra:IsA("BasePart") and pedra)
        if base and base.CanCollide == true then
            IrParaDestino(base.CFrame * CFrame.new(0, 3, 3))
            task.wait(1)
            local camera = workspace.CurrentCamera
            if camera then camera.CFrame = CFrame.lookAt(camera.CFrame.Position, base.Position) end

            local prompt = pedra:FindFirstChildOfClass("ProximityPrompt") or base:FindFirstChildOfClass("ProximityPrompt")
            if prompt then
                PrepararPrompt(prompt)
                local resultado = MinerarPedra(pedra, base, prompt)
                if resultado == "retry" then
                    task.wait(1)
                    continue
                end
            else
                warn("ProximityPrompt da pedra nao encontrado!")
            end
        end

        pedrasJaMineradas[pedra] = true
        task.wait(0.1)
    end
    return false
end
local function VenderMinerios()
    if not InventarioEstaCheio() then
        warn("Peso abaixo de 81; venda cancelada.")
        return false
    end
    local carroPego = false
    while rodando and not carroPego do
        carroPego = PegarCarroETeleportar(CONFIG.WP_VENDEDOR)
        if not carroPego then
            warn("Falhou ao usar carro, tentando novamente...")
            task.wait(2)
        end
    end
    if not rodando then return end
    
    
    -- Usa APENAS o ProximityPrompt do vendedor.
    -- NAO aperta tecla (SendKeyEvent) e NAO procura prompts por perto,
    -- senao ativa o prompt do banco do carro e senta no banco do passageiro.
    local promptAlvo = nil
    pcall(function()
        promptAlvo = workspace.Construcoes.Minerador.Minerador.ProximityBlock.ProximityPrompt
    end)

    if promptAlvo then
        PrepararPrompt(promptAlvo)
        pcall(function() fireproximityprompt(promptAlvo) end)
        pcall(function() promptAlvo:InputHoldBegan() end)
        task.wait(1.5) -- Espera GUI abrir
        pcall(function() promptAlvo:InputHoldEnded() end)
    else
        warn("ProximityPrompt do Vendedor nao encontrado!")
    end

    -- Garantia: se mesmo assim sentou em algum banco, levanta
    if Hum.Sit then PularDoCarro() end
    
    -- Agora acha a UI e vende
    local gui = LocalPlayer.PlayerGui:FindFirstChild("Sistemas")
    gui = gui and gui:FindFirstChild("VenderMinerios")
    
    if not gui then
        warn("⚠️ Menu 'VenderMinerios' não encontrado na PlayerGui!")
        return false
    end
    
    local mineriosAlvo = {"Bronze", "Ferro", "Ouro", "Diamante", "Meteorito"}
    local inventario = LocalPlayer:FindFirstChild("Inventario")
    local itensParaVender = {}
    
    if inventario then
        for _, obj in ipairs(inventario:GetChildren()) do
            table.insert(itensParaVender, obj.Name)
        end
    else
        itensParaVender = mineriosAlvo -- Fallback
    end
    
    for _, minerioName in ipairs(mineriosAlvo) do
        local temNoInventario = false
        if inventario then
            for _, invItem in ipairs(itensParaVender) do
                if invItem == minerioName or (minerioName == "Ferro" and invItem == "Prata") then
                    temNoInventario = true
                    break
                end
            end
        else
            temNoInventario = true
        end
        
        if not temNoInventario then
            continue
        end

        local slot = gui:FindFirstChild(minerioName)
        
        -- Regra especial para Ferro/Prata
        if not slot and minerioName == "Ferro" then
            slot = gui:FindFirstChild("Prata")
        end
        
        if slot then
            local botaoVender = slot:FindFirstChild("vender") or slot:FindFirstChildWhichIsA("TextButton", true) or slot:FindFirstChildWhichIsA("ImageButton", true)
            if botaoVender then
                SimularCliqueUI(botaoVender)
                task.wait(0.2)
                SimularCliqueUI(botaoVender)
                
                -- O Ouro requer que clique mais devagar para nao bugar
                if minerioName == "Ouro" then
                    task.wait(2.0)
                else
                    task.wait(0.5)
                end
            else
                warn(("   ⚠️ Botão 'vender' não encontrado em %s"):format(slot.Name))
            end
        else
            warn(("   ⚠️ Slot de minério '%s' não encontrado no menu"):format(minerioName))
        end
    end
    
    -- Fecha GUI de venda, se possível (alguns fecham desativando Enabled)
    pcall(function() gui.Enabled = false end)
    task.wait(2.5) -- espera sincronizar antes de voltar pra mina
end

local function LoopMinerador()
    
    -- 1. Checa fome/sede antes de tudo
    ChecarEResolverFomeSede()
    
    -- 2. Pega Carro e vai pro local do Minerador
    local carroPego = false
    while rodando and not carroPego do
        carroPego = PegarCarroETeleportar(CONFIG.WP_MINERADOR)
        if not carroPego then
            warn(" Falhou ao usar carro, tentando novamente...")
            task.wait(2)
        end
    end
    if not rodando then return end
    -- (A picareta será pega automaticamente sob demanda na hora de minerar)
    
    -- 4. Minerar todas as pedras possíveis
    MinerarTodasAsPedras()
    
    -- 5. Vender tudo
    if not rodando or not InventarioEstaCheio() then
        return
    end
    VenderMinerios()

    task.wait(1)
end

        end

            if driveSeat and not driveSeat.Occupant then
                pcall(function() driveSeat:Sit(hum) end)
                task.wait(0.3)
                if hum.Sit then return true end
            end
            warn("[Auto Uber] ProximityPrompt do assento nao encontrado!")
            return false
        end

        for i = 1, 10 do
            if not rodando then return false end
            FirePrompt(prompt)
            task.wait(0.1)
            if hum.Sit then
                return true
            end
        end
        
        warn("[Auto Uber] Nao conseguiu entrar no carro")
        return false
    end

    -- Ciclo completo de ir e entrar no carro identico ao Auto Minerador
    local function irParaCarroEEntrar()
        local char = LocalPlayer.Character
        local hum = char and char:FindFirstChildOfClass("Humanoid")
        if not hum then return false end
        
        local veiculo = acharCarro()
        if not veiculo then
            task.wait(0.5)
            veiculo = acharCarro()
            if not veiculo then
                warn("[Auto Uber] Nenhum veiculo encontrado em CarrosSpawnados!")
                return false
            end
        end
        
        -- Se ja esta sentado dentro do veiculo
        local seatAtual = hum.SeatPart
        if hum.Sit and seatAtual and seatAtual:IsDescendantOf(veiculo) then
            return true
        end
        
        if API.setStatus then pcall(API.setStatus, "Indo para o veiculo...") end
        
        if not AndarAteCarro(veiculo) then return false end
        if not EntrarNoCarro(veiculo) then return false end
        
        task.wait(0.5) -- espera o assento sincronizar antes do TP (exatamente como no auto minerador)
        return hum.Sit
    end

    -- Simula o pulo no PC (Spacebar) e no Mobile (Botao de pulo / TouchGui / VirtualUser)
    local function PularDoCarro(vehicle)
        local char = LocalPlayer.Character
        local hum = char and char:FindFirstChildOfClass("Humanoid")
        local hrp = char and char:FindFirstChild("HumanoidRootPart")
        if not hum or not hrp then return false end

        for tentativa = 1, 10 do
            -- 1. Pulo nativo do Humanoid
            pcall(function()
                hum.Jump = true
                hum:ChangeState(Enum.HumanoidStateType.Jumping)
            end)

            -- 2. Teclado via VirtualInputManager (Barra de Espaço)
            pcall(function()
                local vim = game:GetService("VirtualInputManager")
                vim:SendKeyEvent(true, Enum.KeyCode.Space, false, game)
                task.wait(0.04)
                vim:SendKeyEvent(false, Enum.KeyCode.Space, false, game)
            end)

            -- 3. Mobile: Clica no JumpButton do TouchGui se existir
            pcall(function()
                local playerGui = LocalPlayer:FindFirstChildOfClass("PlayerGui") or LocalPlayer:FindFirstChild("PlayerGui")
                if playerGui then
                    local touchGui = playerGui:FindFirstChild("TouchGui")
                    local touchControlFrame = touchGui and touchGui:FindFirstChild("TouchControlFrame")
                    local jumpButton = touchControlFrame and touchControlFrame:FindFirstChild("JumpButton")
                    if jumpButton and jumpButton:IsA("GuiButton") then
                        if getconnections then
                            for _, conn in ipairs(getconnections(jumpButton.Activated)) do conn:Fire() end
                            for _, conn in ipairs(getconnections(jumpButton.MouseButton1Down)) do conn:Fire() end
                            for _, conn in ipairs(getconnections(jumpButton.MouseButton1Click)) do conn:Fire() end
                        end
                        -- Clique virtual nas coordenadas do botao de pulo na tela
                        local pos = jumpButton.AbsolutePosition + (jumpButton.AbsoluteSize / 2)
                        local vim = game:GetService("VirtualInputManager")
                        vim:SendTouchEvent(1, 0, pos.X, pos.Y)
                        task.wait(0.04)
                        vim:SendTouchEvent(1, 2, pos.X, pos.Y)
                    end
                end
            end)

            -- 4. Mobile via VirtualUser (Input de Controle/Salto)
            pcall(function()
                local vu = game:GetService("VirtualUser")
                vu:CaptureController()
                vu:SetKeyDown("0x20") -- Space / Jump code
                task.wait(0.04)
                vu:SetKeyUp("0x20")
            end)

            task.wait(0.15)

            -- Se desceu/pulou do veiculo
            if not hum.Sit then
                local pPart = vehicle and (vehicle.PrimaryPart or vehicle:FindFirstChildWhichIsA("BasePart"))
                if pPart then
                    pcall(function()
                        hrp.CFrame = pPart.CFrame * CFrame.new(8, 3, 0)
                        hrp.AssemblyLinearVelocity = Vector3.zero
                    end)
                end
                task.wait(0.3)
                if not hum.Sit then
                    return true
                end
            end
        end
        return not hum.Sit
    end

    -- Teleporta o carro direto em 1 vez sem passos intermediarios (sem ir de 41, 42... 50)
    local function TeleportarCarro(veiculo, destinoCFrame)
        if not veiculo then return false, false end
        local pPart = veiculo.PrimaryPart or veiculo:FindFirstChildWhichIsA("BasePart")
        if not pPart then return false, false end

        if typeof(destinoCFrame) == "Vector3" then
            destinoCFrame = CFrame.new(destinoCFrame)
        end
        local alvoPos = destinoCFrame.Position
        local posInicial = pPart.Position
        local dist = (posInicial - alvoPos).Magnitude
        local foiTp = dist > 20 -- Identifica se foi um teleporte direto de distancia


        protegerCarro(veiculo)
        local deslocamentoDoPivo = pPart.CFrame:ToObjectSpace(veiculo:GetPivot())
        local ok = pcall(function()
            pPart.AssemblyLinearVelocity = Vector3.zero
            pPart.AssemblyAngularVelocity = Vector3.zero
            veiculo:PivotTo(destinoCFrame * deslocamentoDoPivo)
        end)
        if not ok then return false, foiTp end
        pararCarroTotal(veiculo)
        task.wait(0.3)
        if not rodando or not veiculo.Parent or (pPart.Position - alvoPos).Magnitude > 30 then
            return false, foiTp
        end
        return true, foiTp
    end

                return false, false
            end
            veiculo = acharCarro()
        end

        if not veiculo then return false, false end
        return TeleportarCarro(veiculo, destinoCFrame)
    end

    local function uberFuelWait(seconds)
        local deadline = os.clock() + seconds
        repeat
            if not rodando then return false end
            task.wait(math.min(0.05, math.max(0, deadline - os.clock())))
        until os.clock() >= deadline
        return rodando
    end
local function uberFuelPromptAnchor(prompt)
    local parent = prompt and prompt.Parent
    if parent and parent:IsA("BasePart") then return parent.CFrame end
    if parent and parent:IsA("Attachment") then return parent.WorldCFrame end
    return nil
end
local function uberHoldFuelPrompt(prompt)
    if not prompt or not prompt.Parent or not prompt.Enabled then return false end
    local anchor = uberFuelPromptAnchor(prompt)
    local character = LocalPlayer.Character
    local root = character and character:FindFirstChild("HumanoidRootPart")
    if not anchor or not root then return false end

    local standingAt = anchor.Position + anchor.LookVector * 2 + Vector3.new(0, 1, 0)
    root.CFrame = CFrame.lookAt(standingAt, anchor.Position)
    if not uberFuelWait(0.6) then return false end
    if not prompt.Enabled or (root.Position - anchor.Position).Magnitude > prompt.MaxActivationDistance then
        return false
    end

    local camera = workspace.CurrentCamera
    local oldCameraType = camera and camera.CameraType
    local oldCameraCFrame = camera and camera.CFrame
    if camera then
        camera.CameraType = Enum.CameraType.Scriptable
        camera.CFrame = CFrame.lookAt(root.Position + Vector3.new(0, 2, 0), anchor.Position)
    end
    if not uberFuelWait(0.35) then
        if camera then camera.CameraType = oldCameraType; camera.CFrame = oldCameraCFrame end
        return false
    end

    local triggered = false
    local connection = prompt.Triggered:Connect(function()
        triggered = true
    end)
    local held = false
    if type(isrbxactive) == "function" and isrbxactive()
        and type(keypress) == "function" and type(keyrelease) == "function" then
        keypress(0x45)
        held = uberFuelWait(prompt.HoldDuration + 0.3)
        keyrelease(0x45)
    else
        local began = pcall(function() prompt:InputHoldBegin() end)
        held = began and uberFuelWait(prompt.HoldDuration + 0.3)
        if began then pcall(function() prompt:InputHoldEnd() end) end
    end
    uberFuelWait(0.3)
    connection:Disconnect()
    if camera then camera.CameraType = oldCameraType; camera.CFrame = oldCameraCFrame end
    return held and triggered
end
local function AbastecerVeiculo(veiculo)
    local vehicle = acharCarro() or veiculo
    if not vehicle then return false end
    local atStation = (vehicle:GetPivot().Position - cframePostoGasolina.Position).Magnitude < 35
    if not atStation then
        local humanoid = LocalPlayer.Character and LocalPlayer.Character:FindFirstChildOfClass("Humanoid")
        if not humanoid or not humanoid.SeatPart then
            if not irParaCarroEEntrar() then return false end
        end
    end
    local info = vehicle and vehicle:FindFirstChild("Info")
    local status = info and info:FindFirstChild("Status")
    local gas = status and status:FindFirstChild("Gasolina")
    if not gas or not gas:IsA("NumberValue") then return false end

    setStatus("Gasolina baixa: indo ao posto")
    local constructions = workspace:FindFirstChild("Construcoes")
    local station = constructions and (constructions:FindFirstChild("PostoDe Gasolina")
        or constructions:FindFirstChild("PostoDeGasolina"))
    local pumps = station and station:FindFirstChild("Bombas")
    if not pumps then return false end
    -- As bombas podem estar vazias pelo StreamingEnabled enquanto o jogador está longe.
    -- Primeiro chega ao posto com o carro; só então escolhe um engate carregado.
    if not atStation and not TeleportarCarro(vehicle, cframePostoGasolina) then
        setStatus("Nao foi possivel chegar ao posto")
        return false
    end
    setStatus("Aguardando a bomba carregar")
    local pump, nozzlePrompt
    local deadline = os.clock() + 12
    repeat
        local nearest = math.huge
        for _, candidate in ipairs(pumps:GetChildren()) do
            if candidate.Name == "BombaGasolina" then
                local nozzle = candidate:FindFirstChild("engate", true)
                local prompt = nozzle and (nozzle:IsA("ProximityPrompt") and nozzle
                    or nozzle:FindFirstChildOfClass("ProximityPrompt"))
                if prompt and prompt.Enabled then
                    local anchor = uberFuelPromptAnchor(prompt)
                    local distance = anchor and (anchor.Position - vehicle:GetPivot().Position).Magnitude or math.huge
                    if distance < nearest then
                        pump, nozzlePrompt, nearest = candidate, prompt, distance
                    end
                end
            end
        end
        if nozzlePrompt then break end
        if not uberFuelWait(0.5) then return false end
    until os.clock() >= deadline
    if not nozzlePrompt then
        setStatus("Engate da bomba nao carregou")
        return false
    end

    local character = LocalPlayer.Character
    local humanoid = character and character:FindFirstChildOfClass("Humanoid")
    if not humanoid then return false end
    if humanoid.SeatPart then
        humanoid.Sit = false
        humanoid:ChangeState(Enum.HumanoidStateType.GettingUp)
        if not uberFuelWait(0.6) then return false end
    end
    if humanoid.SeatPart then return false end

    setStatus("Segurando o prompt da bomba")
    if not uberHoldFuelPrompt(nozzlePrompt) then
        setStatus("Bomba nao confirmou o prompt")
        return false
    end
    if not uberFuelWait(1.5) then return false end

    local body = vehicle:FindFirstChild("Body")
    local gasPart = body and body:FindFirstChild("Gasolina")
    local proxy = gasPart and gasPart:FindFirstChild("GasolinaProxy")
    local tankPrompt = proxy and (proxy:IsA("ProximityPrompt") and proxy
        or proxy:FindFirstChildOfClass("ProximityPrompt"))
    tankPrompt = tankPrompt or (gasPart and gasPart:FindFirstChildOfClass("ProximityPrompt"))
    if not tankPrompt then
        setStatus("Prompt do tanque nao encontrado")
        return false
    end
    if humanoid.SeatPart then
        humanoid.Sit = false
        humanoid:ChangeState(Enum.HumanoidStateType.GettingUp)
        if not uberFuelWait(0.6) or humanoid.SeatPart then return false end
    end
    setStatus("Segurando o prompt do tanque")
    if not uberHoldFuelPrompt(tankPrompt) then
        setStatus("Tanque nao confirmou o prompt")
        return false
    end

    setStatus("Aguardando o tanque encher")
    local deadline = os.clock() + 45
    while rodando and os.clock() < deadline do
        if gas.Value >= 95 then
            setStatus("Abastecimento confirmado")
            return irParaCarroEEntrar()
        end
        if not uberFuelWait(0.5) then return false end
    end
    return false
end

    local function LoopUber()
        ativarNoclip()
        
        local checkpointsFeitos = {}
        local ultimoCheckpointFeito = 0
        local ultimoCheckpointPos = nil
        local ultimoAceitarTick = 0
        
        while rodando do
            local foodOk, foodError = pcall(ChecarEResolverFomeSede)
            if not foodOk then setStatus("Auto comer: " .. tostring(foodError)) end
            
            local hrp = LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("HumanoidRootPart")
            local hum = LocalPlayer.Character and LocalPlayer.Character:FindFirstChildOfClass("Humanoid")
            if not hrp or not hum then
                task.wait(1)
                continue
            end

            -- 0. Monitoramento e Abastecimento de Gasolina
            local meuCarro = acharCarro()
            protegerCarro(meuCarro)
            local nivelGas = obterGasolina(meuCarro)
            if nivelGas and nivelGas <= 10 then
                if not AbastecerVeiculo(meuCarro) then
                    setStatus("Abastecimento pendente; tentando novamente")
                    task.wait(3)
                    continue
                end
            end
            
            local emRotaAtiva = checarEmRota()
            
            -- 1. So aceita corrida se NAO estiver em rota ativa
            if not emRotaAtiva then
                if tick() - ultimoAceitarTick > 2.5 then
                    ultimoAceitarTick = tick()
                    aceitarCorrida()
                end
            end
            
            -- Se tiver corrida aceita / ativa:
            if checarEmRota() then
                -- 2. Entra no veiculo
                local humAtual = LocalPlayer.Character and LocalPlayer.Character:FindFirstChildOfClass("Humanoid")
                if not humAtual or not humAtual.Sit then
                    setStatus("Indo para o veiculo...")
                    local entrou = irParaCarroEEntrar()
                    if not entrou then
                        warn("[Auto Uber] Tentou entrar no veiculo mas nao conseguiu sentar, aguardando...")
                    end
                    task.wait(0.3)
                end
                
                -- LocalMarcado tambem aponta para os checkpoints depois do embarque.
                local uberState = LocalPlayer:FindFirstChild("Uber")
                local estadoPassageiro = uberState and uberState:FindFirstChild("InicialPassageiro")
                local passageiroEmbarcou = estadoPassageiro and estadoPassageiro.Value

                -- 3.1 Busca o cliente somente enquanto o jogo ainda nao confirmou o embarque.
                local destinoMarcado = obterMarcadorUber()
                if destinoMarcado and not passageiroEmbarcou then
                    setStatus("Indo buscar passageiro...")
                    
                    local posTarget = destinoMarcado:IsA("BasePart") and destinoMarcado.CFrame
                        or (destinoMarcado:IsA("Attachment") and CFrame.new(destinoMarcado.WorldPosition))
                        or destinoMarcado:GetPivot()
                    
                    if posTarget then
                        moverPara(posTarget)
                        local meuCarro = acharCarro()
                        pararCarroTotal(meuCarro)
                    end
                    
                    -- O jogo confirma o embarque em InicialPassageiro; sem espera artificial.
                    passageiroEmbarcou = estadoPassageiro and estadoPassageiro.Value
                    if passageiroEmbarcou then
                        setStatus("Passageiro a bordo!")
                        checkpointsFeitos = {}
                        ultimoCheckpointFeito = 0
                        ultimoCheckpointPos = nil
                    end
                end
                
                -- 3.2 O marcador continua presente durante a rota; o estado do jogo decide a etapa.
                if passageiroEmbarcou then
                    local construcoes = workspace:FindFirstChild("Construcoes")
                    local locais = construcoes and construcoes:FindFirstChild("LocaisCorrida")
                    local rotasFolder = locais and locais:FindFirstChild("RotasCliente")
                    local pontoAtual = nil
                    
                    if rotasFolder then
                        local lista = {}
                        for idxChild, item in ipairs(rotasFolder:GetChildren()) do
                            local num = tonumber(item.Name:match("%d+")) or tonumber(item.Name) or idxChild
                            local cf = item:IsA("BasePart") and item.CFrame or (item:IsA("Model") and item:GetPivot() or nil)
                            if cf then
                                local posKey = ("%.0f_%.0f_%.0f"):format(math.floor(cf.Position.X / 10) * 10, math.floor(cf.Position.Y / 10) * 10, math.floor(cf.Position.Z / 10) * 10)
                                table.insert(lista, {index = num, pos = cf, obj = item, key = posKey})
                            end
                        end
                        -- A pasta pode conter pontos antigos; o marcador mostra o alvo atual.
                        local marcadores = workspace:FindFirstChild("Marcadores")
                        local pastaUber = marcadores and marcadores:FindFirstChild("Uber")
                        local marcadorAtual = pastaUber and pastaUber:FindFirstChild("LocalMarcado")
                        local marcadorPos = marcadorAtual and (marcadorAtual:IsA("BasePart") and marcadorAtual.Position
                            or (marcadorAtual:IsA("Attachment") and marcadorAtual.WorldPosition)
                            or marcadorAtual:GetPivot().Position)
                        if marcadorPos then
                            local menorDist = 8
                            for _, cp in ipairs(lista) do
                                local dist = (cp.pos.Position - marcadorPos).Magnitude
                                if dist < menorDist then
                                    pontoAtual = cp
                                    menorDist = dist
                                end
                            end
                        end
                    end
                    
                    if pontoAtual then
                        -- Garante que o player esteja sentado no veiculo antes de qualquer movimento
                        local humCheck = LocalPlayer.Character and LocalPlayer.Character:FindFirstChildOfClass("Humanoid")
                        if not humCheck or not humCheck.Sit then
                            irParaCarroEEntrar()
                        end
                        
                        rotaCheckRodou = false
                        
                        local meuCarro = acharCarro()

                        -- Vai diretamente ao checkpoint atual, sem destino intermediario.
                        setStatus(("No Checkpoint %d"):format(pontoAtual.index))
                        moverPara(pontoAtual.pos)
                        
                        pararCarroTotal(meuCarro)
                        local tPonto = tick()
                        while rodando and tick() - tPonto < 0.6 do
                            pararCarroTotal(meuCarro)
                            task.wait(0.05)
                        end
                        pararCarroTotal(meuCarro)
                        
                        -- 3. Verificacao se o servidor ja contabilizou ou rodou o remote:
                        local jaContabilizou = false
                        if rotaCheckRodou then jaContabilizou = true end
                        if pontoAtual.obj and (not pontoAtual.obj.Parent or pontoAtual.obj.Parent ~= rotasFolder) then
                            jaContabilizou = true
                        end
                        if not checarEmRota() then jaContabilizou = true end
                        
                        -- Se teleportou e o jogo nao rodou o remote nem contabilizou: envia o remote!
                        if not jaContabilizou then
                            invocarRotaCheck()
                            task.wait(0.2)
                        end
                        
                        -- 4. Marca este checkpoint como concluido definitivamente
                        checkpointsFeitos[pontoAtual.index] = true
                        if pontoAtual.obj then checkpointsFeitos[pontoAtual.obj] = true end
                        checkpointsFeitos[pontoAtual.key] = true
                        ultimoCheckpointPos = pontoAtual.pos.Position
                        ultimoCheckpointFeito = math.max(ultimoCheckpointFeito, pontoAtual.index)
                        
                        -- Mantem o tempo configurado entre checkpoints.
                        local tempoEspera = math.max((API.WaitTime or 1) - 1.5, 1)
                        task.wait(tempoEspera)
                    else
                        if not checarEmRota() then
                            setStatus("Corrida Finalizada!")
                            checkpointsFeitos = {}
                            ultimoCheckpointPos = nil
                            ultimoCheckpointFeito = 0
                            task.wait(0.1)
                        else
                            setStatus("Aguardando proximo ponto...")
                            task.wait(0.5)
                        end
                    end
                end
            else
                setStatus("Aguardando Corrida...")
                task.wait(0.15)
            end
            
            task.wait(0.1)
        end
        
        desativarNoclip()
        pararProtecaoCarro()
        setStatus("Parado")
    end

    function API.iniciar()
        if rodando then return false end
        rodando = true

        -- Define o caminho do RemoteFunction uma vez para melhorar o desempenho e roda a cada 3s
        Runtime:RunUnique("UberFreioDeMao", function()
            local assentoAnterior = nil
            while rodando and Runtime.Active do
                local personagem = LocalPlayer.Character
                local humanoide = personagem and personagem:FindFirstChildOfClass("Humanoid")
                local assento = humanoide and humanoide.SeatPart
                if assento and assento ~= assentoAnterior and assento:IsA("VehicleSeat") then
                    if type(isrbxactive) == "function" and isrbxactive()
                        and type(keypress) == "function" and type(keyrelease) == "function" then
                        keypress(0x10) -- Shift
                        task.wait(0.07)
                        keyrelease(0x10)
                        setStatus("Freio de mao acionado")
                    else
                        pcall(function()
                            local vim = game:GetService("VirtualInputManager")
                            vim:SendKeyEvent(true, Enum.KeyCode.LeftShift, false, game)
                            task.wait(0.07)
                            vim:SendKeyEvent(false, Enum.KeyCode.LeftShift, false, game)
                        end)
                    end
                end
                assentoAnterior = assento
                task.wait(0.15)
            end
        end)

        Runtime:RunUnique("UberRotaCheck", function()
            local ReplicatedStorage = game:GetService("ReplicatedStorage")
            local RotaCheck = nil
            pcall(function()
                local uberFolder = ReplicatedStorage:WaitForChild("Uber", 10)
                RotaCheck = uberFolder and uberFolder:WaitForChild("RotaCheck", 10)
            end)

            if not RotaCheck then
                warn("[Auto Uber] RemoteFunction RotaCheck nao encontrado!")
                return
            end

            while rodando do
                pcall(function()
                    RotaCheck:InvokeServer()
                end)
                task.wait(3)
            end
        end)

        Runtime:RunUnique("Uber", function()
            local s, e = pcall(LoopUber)
            if not s then warn("[Auto Uber] Erro: " .. tostring(e)) end
            rodando = false
            desativarNoclip()
            pararProtecaoCarro()
        end)
        return true
    end

    function API.parar()
        rodando = false
        desativarNoclip()
        pararProtecaoCarro()
    end

    function API.isRunning()
        return rodando
    end

    env.KLZAutoUberAPI = API
    return API
end)()

-- Levanta automaticamente quando a vida chega a 1 ou 0 e o contador libera o botao.
local function botaoLevantarDisponivel()
    local character = LocalPlayer.Character
    local humanoid = character and character:FindFirstChildOfClass("Humanoid")
    if not humanoid or humanoid.Health > 1 then return nil end

    local respawn = LocalPlayer.PlayerGui:FindFirstChild("Respawn")
    local morte = respawn and respawn:FindFirstChild("Morte")
    local main = morte and morte:FindFirstChild("Main")
    local frame = main and main:FindFirstChild("RespawnButton")
    local button = frame and frame:FindFirstChild("Button")
    local contador = main and main:FindFirstChild("cont")
    if not respawn or not respawn.Enabled or not morte.Visible then return nil end
    if not button or not button:IsA("GuiButton") or not button.Visible or not button.Active then return nil end
    if not contador then return nil end

    local segundos = tonumber(contador.Text:match("(%d+)%s*segundos"))
    if segundos == nil or segundos > 0 then return nil end
    return button, respawn
end

local function clicarLevantar(button, respawn)
    local centro = button.AbsolutePosition + button.AbsoluteSize / 2
    local inset = GuiService:GetGuiInset()
    local x = math.floor(centro.X)
    local y = math.floor(centro.Y + (respawn.IgnoreGuiInset and inset.Y or 0))

    if type(isrbxactive) == "function" and isrbxactive()
        and type(mousemoveabs) == "function" and type(mouse1click) == "function" then
        local ok = pcall(function()
            mousemoveabs(x, y)
            mouse1click()
        end)
        if ok then return end
    end

    pcall(function()
        VirtualInputManager:SendMouseButtonEvent(x, y, 0, true, game, 0)
        task.wait(0.06)
        VirtualInputManager:SendMouseButtonEvent(x, y, 0, false, game, 0)
    end)
end

Runtime:RunUnique("KLZAutoLevantar", function()
    local ultimaTentativa = 0
    while Runtime.Active do
        local button, respawn = botaoLevantarDisponivel()
        if button and os.clock() - ultimaTentativa >= 2 then
            ultimaTentativa = os.clock()
            clicarLevantar(button, respawn)
        end
        task.wait(0.25)
    end
end)

local enabled = false
local minerEnabled = false
local setMinerEnabled
local function setEnabled(value)
    if enabled == value then return end
    if value and minerEnabled then setMinerEnabled(false) end
    enabled = value
    if enabled then
        startAntiFly()
        if not AutoUberAPI.iniciar() then
            enabled = false
            stopAntiFly()
        end
    else
        AutoUberAPI.parar()
        stopAntiFly()
    end
    toggle.Text = enabled and "AUTO UBER  ·  ON" or "AUTO UBER  ·  OFF"
    toggle.BackgroundColor3 = enabled and Color3.fromRGB(55, 103, 77) or Color3.fromRGB(45, 47, 52)
    if not enabled then status.Text = "Parado" end
end
Runtime:TrackConnection(toggle.Activated:Connect(function() setEnabled(not enabled) end))

setMinerEnabled = function(value)
    if minerEnabled == value then return end
    if value and enabled then setEnabled(false) end
    minerEnabled = value
    if minerEnabled then
        startAntiFly()
        if not AutoMinerador.iniciar() then
            minerEnabled = false
            stopAntiFly()
        end
    else
        AutoMinerador.parar()
        stopAntiFly()
    end
    minerToggle.Text = minerEnabled and "AUTO MINERADOR  ·  ON" or "AUTO MINERADOR  ·  OFF"
    minerToggle.BackgroundColor3 = minerEnabled and Color3.fromRGB(55, 103, 77) or Color3.fromRGB(45, 47, 52)
    minerStatus.Text = minerEnabled and "Minerador: Rodando" or "Minerador: Parado"
end
Runtime:TrackConnection(minerToggle.Activated:Connect(function() setMinerEnabled(not minerEnabled) end))

local function stop()
    setEnabled(false)
    setMinerEnabled(false)
    Runtime.Active = false
    for _, conn in ipairs(Runtime.Connections) do pcall(function() conn:Disconnect() end) end
    for _, tween in ipairs(Runtime.Tweens) do pcall(function() tween:Cancel() end) end
    for _, instance in ipairs(Runtime.Instances) do pcall(function() instance:Destroy() end) end
    for _, cleanup in ipairs(Runtime.Cleanups) do pcall(cleanup) end
    for _, thread in pairs(Runtime.Workers) do pcall(task.cancel, thread) end
    screen:Destroy()
    if env.KLZBypasse and env.KLZBypasse.Stop == stop then env.KLZBypasse = nil end
    if env.KLZAutoUberAPI == AutoUberAPI then env.KLZAutoUberAPI = nil end
    if env.KLZAutoMineradorAPI == AutoMinerador then env.KLZAutoMineradorAPI = nil end
end
env.KLZBypasse = {Stop = stop, SetEnabled = setEnabled, SetMinerEnabled = setMinerEnabled,
